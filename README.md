## 📌 Table Creation – Contact
```sql
CREATE TABLE contact (
  eid INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  name VARCHAR(50),
  email VARCHAR(50),
  mob VARCHAR(10),
  age INT,
  city VARCHAR(50)

```

## 📌 WHERE Clause Examples
```sql
SELECT * FROM contact WHERE city='Raipur';
SELECT * FROM contact WHERE age < 19;
SELECT * FROM contact WHERE age > 19;
SELECT * FROM contact WHERE age = 19;
```
### AND / OR
```sql
SELECT * FROM contact WHERE age <= 19 AND city='Raipur';      Dono conditions true honi chahiye.
SELECT * FROM contact WHERE age <= 19 OR city='Raipur';       Koi bhi ek condition true ho to row aa jayegi.
```
### IN / NOT IN
```sql
SELECT * FROM contact WHERE city IN ('Mahasamund','Raipur');   Jinki city Mahasamund ya Raipur ho, unhe select karo.
SELECT * FROM contact WHERE city NOT IN ('kMahasamund','Raipur');  Jinki city Mahasamund aur Raipur ke alawa koi aur ho, unhe select karo.
```
## 📌 ORDER BY

### Salary Queries {Aggregate Functions}
```sql
SELECT MAX(salary) FROM contact;  Highest Salary
SELECT MIN(salary) FROM contact;  Lowest Salary
SELECT AVG(salary::numeric) FROM contact;  Average Salary
SELECT SUM(salary::numeric) FROM contact;  Total Salary
SELECT COUNT(*) FROM contact;  Total Employees
Highest / Lowest Salary Employee
SELECT name, salary FROM contact WHERE salary = (SELECT MAX(salary) FROM contact);  Highest Salary Employee
SELECT name, salary FROM contact WHERE salary = (SELECT MIN(salary) FROM contact);  Lowest Salary Employee

```
## 📌 LIMIT & OFFSET
```sql
SELECT name, salary FROM contact ORDER BY salary DESC LIMIT 3;  Top 3 Highest Salary Employee
SELECT name, salary FROM contact ORDER BY salary DESC LIMIT 1 OFFSET 1; Second Highest Salary employee
```

## INNER JOIN सबसे ज़्यादा उपयोग होता है केवल वही records आएंगे जिनका gid दोनों tables में match करता है।
```SQL  
SELECT menu_name, menu_price, group_name
FROM menu
INNER JOIN food_group
ON menu.gid = food_group.gid;
```
## LEFT JOIN  menu table के सभी records आएंगे।
यदि food_group में match नहीं मिला तो group_name में NULL आएगा।
```SQL
SELECT menu_name, menu_price, group_name
FROM menu
LEFT JOIN food_group
ON menu.gid = food_group.gid;
```
## RIGHT JOIN food_group table के सभी records आएंगे।
यदि menu में match नहीं मिला तो menu_name और menu_price NULL होंगे।
```SQL
SELECT menu_name, menu_price, group_name
FROM menu
RIGHT JOIN food_group
ON menu.gid = food_group.gid;
```
## CROSS JOIN दोनों tables का हर record एक-दूसरे के साथ combine होगा।
यदि menu में 5 rows और food_group में 3 rows हैं, तो output में 15 rows आएंगी।
```SQL
SELECT menu_name, menu_price, group_name
FROM menu
CROSS JOIN food_group;
```
## UNION यह food_group table के group_name और qtymast table के qty_type को एक ही result में जोड़ देता है।
```sql
SELECT group_name FROM food_group
UNION
SELECT qty_type FROM qtymast;
```

### Create a function Backup Before Delete
```sql
CREATE OR REPLACE FUNCTION backup()
RETURNS TRIGGER AS $$
BEGIN
  INSERT INTO back_contact(emp_id,name, email, mobile, age, city, salary, department)
  VALUES (OLD.eid, OLD.name, OLD.email, OLD.mobile, OLD.age, OLD.city, OLD.salary, OLD.department);

  RETURN OLD;
END;
$$ LANGUAGE plpgsql;
```
## 📌 Create Triggers (Backup)
```sql
CREATE TRIGGER backup_trigger
BEFORE DELETE
ON contact
FOR EACH ROW
EXECUTE FUNCTION backup();
```
## View 

### Create a function Audit After Insert
```sql
CREATE OR REPLACE FUNCTION auditlog()
RETURNS TRIGGER AS $$
BEGIN
INSERT INTO audit(emp_id, entry_date)
VALUES (NEW.eid, CURRENT_TIMESTAMP);
RETURN NEW;
END;
$$ LANGUAGE plpgsql;
```
## Create Triggers (Backup & Audit)
```sql
CREATE TRIGGER audit_trigger
AFTER INSERT
ON contact
FOR EACH ROW
```
